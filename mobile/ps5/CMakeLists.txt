cmake_minimum_required(VERSION 3.20)
project(BruTV VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BRUTV_PS5 "Build a native PlayStation 5 homebrew title" OFF)
option(BRUTV_BUILD_APP "Build the OpenGL application" ON)

if(BRUTV_PS5)
  include(cmake/PS5Dependencies.cmake)
  set(BRUTV_ASSET_DIR "/app0/assets")
  add_library(brutv_json INTERFACE)
  target_include_directories(brutv_json INTERFACE "${BRUTV_JSON_INCLUDE_DIR}")
else()
  find_package(nlohmann_json 3.11 REQUIRED)
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
  find_package(CURL REQUIRED)
  add_library(brutv_json ALIAS nlohmann_json::nlohmann_json)
  set(BRUTV_ASSET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/assets")
endif()

add_library(brutv_catalog STATIC src/catalog.cpp)
target_include_directories(brutv_catalog PUBLIC include)
target_link_libraries(brutv_catalog PUBLIC brutv_json)
add_library(brutv_player STATIC src/player.cpp)
target_include_directories(brutv_player PUBLIC include)
add_library(brutv_artwork STATIC src/artwork.cpp)
target_include_directories(brutv_artwork PUBLIC include)
target_include_directories(brutv_artwork SYSTEM PRIVATE third_party)
if(BRUTV_PS5)
  target_compile_definitions(brutv_catalog PRIVATE BRUTV_PS5=1)
  target_compile_definitions(brutv_player PRIVATE BRUTV_PS5=1)
  target_include_directories(brutv_player PRIVATE ${PS5_MEDIA_INCLUDE_DIRS})
else()
  target_link_libraries(brutv_player PUBLIC PkgConfig::FFMPEG Threads::Threads)
  target_link_libraries(brutv_artwork PUBLIC Threads::Threads)
endif()

if(BRUTV_BUILD_APP)
  add_executable(brutv src/main.cpp src/app.cpp src/gfx.cpp src/http.cpp)
  target_include_directories(brutv PRIVATE include)
  target_include_directories(brutv SYSTEM PRIVATE third_party)
  target_compile_definitions(brutv PRIVATE GL_GLEXT_PROTOTYPES=1 BRUTV_ASSET_DIR="${BRUTV_ASSET_DIR}")
  target_link_libraries(brutv PRIVATE brutv_catalog brutv_player brutv_artwork)
  if(BRUTV_PS5)
    target_sources(brutv PRIVATE src/platform_ps5.cpp)
    target_link_libraries(brutv PRIVATE brutv_ps5_platform)
  else()
    find_package(SDL2 REQUIRED)
    find_package(OpenGL REQUIRED)
    target_sources(brutv PRIVATE src/platform_sdl.cpp)
    target_link_libraries(brutv PRIVATE SDL2::SDL2 OpenGL::GL CURL::libcurl)
  endif()
endif()

if(NOT BRUTV_PS5)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(brutv_catalog_test tests/catalog_test.cpp)
    target_link_libraries(brutv_catalog_test PRIVATE brutv_catalog)
    target_compile_options(brutv_catalog_test PRIVATE -UNDEBUG)
    add_test(NAME catalog COMMAND brutv_catalog_test)
    add_executable(brutv_artwork_test tests/artwork_test.cpp)
    target_include_directories(brutv_artwork_test SYSTEM PRIVATE third_party)
    target_link_libraries(brutv_artwork_test PRIVATE brutv_artwork)
    add_test(NAME artwork COMMAND brutv_artwork_test)
    if(BRUTV_BUILD_APP)
      add_executable(brutv_app_test tests/app_test.cpp src/app.cpp src/gfx.cpp src/platform_sdl.cpp)
      target_include_directories(brutv_app_test PRIVATE include)
      target_include_directories(brutv_app_test SYSTEM PRIVATE third_party)
      target_compile_definitions(brutv_app_test PRIVATE GL_GLEXT_PROTOTYPES=1 BRUTV_TEST_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
      target_link_libraries(brutv_app_test PRIVATE brutv_catalog brutv_player brutv_artwork SDL2::SDL2 OpenGL::GL)
      find_program(XVFB_RUN xvfb-run REQUIRED)
      add_test(NAME controller_source_setup COMMAND "${XVFB_RUN}" -a env SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1 $<TARGET_FILE:brutv_app_test>)
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/player_test.cpp")
      add_executable(brutv_player_test tests/player_test.cpp)
      target_link_libraries(brutv_player_test PRIVATE brutv_player)
    endif()
  endif()
endif()
